+def _evict_failed_native_result(module, cache_name: str) -> None:
+    """The V10/V11 loaders cache a FAILED verification for the life of the process. A failure observed at worker start
+    must not become permanent where the first Run would have verified afresh, so only failed entries are removed
+    (under the loader's own lock); a successful result is never touched and nothing is made admissible by this."""
+    lock, cache = getattr(module, "_LOCK", None), getattr(module, cache_name, None)
+    if lock is None or not isinstance(cache, dict):
+        return
+    with lock:
+        for key in [k for k, v in cache.items() if isinstance(v, tuple) and v and v[0] is None]:
+            cache.pop(key, None)

+def _native_prewarm() -> dict:
+    """Worker-side native readiness, executed in the worker PID that will serve Runs — NON-FINANCIAL.
+
+    Calls ONLY the existing approved loader entry points (``get_native`` / ``get_stage2`` / ``get_stage3``) for the
+    stages the Run-time activation settings would currently enable (kill switches and modes are read exactly as
+    ``native_activation`` reads them). Those loaders perform the library, ABI, build-ID and canonical-source
+    authentication checks, once per process; this moves that first-use cost from the first user Run to worker start.
+    No project is executed and no admission decision is made here: every Run still decides per call. Never raises."""
+    import time as _time
+
+    out: dict = {"configured": False, "ready": False, "stages": {}, "seconds": 0.0}
+    started = _time.perf_counter()
+    try:
+        if os.environ.get(ENV_NATIVE_PREWARM, "").strip().lower() == "off":
+            out["state"] = "disabled"
+            return out
+        from app.runtime import native_activation as na
+
+        v10, _r10 = na.setting()
+        v11, _r11 = na.v11_setting()
+        out["v10_mode"], out["v11_mode"] = v10, v11
+
+        def stage(name, loader_fn, module, cache_name, pin_env):
+            t0 = _time.perf_counter()
+            result, reason = loader_fn()
+            rec = {"ready": result is not None, "reason": reason, "seconds": round(_time.perf_counter() - t0, 4)}
+            if result is not None:
+                build_id = getattr(result, "build_id", None)
+                if build_id is None and isinstance(result, tuple):
+                    build_id = result[1]
+                rec["build_id"] = build_id
+                pin = (os.environ.get(pin_env) or "").strip() or None if pin_env else None
+                if pin and build_id != pin:
+                    rec["ready"], rec["reason"] = False, "BUILD_ID_PIN_MISMATCH"
+            else:
+                _evict_failed_native_result(module, cache_name)
+            out["stages"][name] = rec
+
+        if v10 != "off":
+            from financial_engine.native_v10 import loader
+            stage("v10_stage1", loader.get_native, loader, "_RESULT", na.ENV_BUILD_ID)
+            if v10 == "stage1+2":
+                from financial_engine.native_v10 import stage2
+                stage("v10_stage2", stage2.get_stage2, stage2, "_LIB", None)   # no separate Stage 2 pin exists
+        if v11 != "off":
+            from financial_engine.native_v11 import stage3
+            stage("v11_stage3", stage3.get_stage3, stage3, "_RESULT", na.ENV_V11_BUILD_ID)
+        out["configured"] = bool(out["stages"])
+        out["ready"] = out["configured"] and all(rec["ready"] for rec in out["stages"].values())
+        out["state"] = ("not_configured" if not out["configured"] else "complete" if out["ready"] else "failed")
+    except BaseException as exc:  # noqa: BLE001 - a prewarm problem must never break worker start
+        out["ready"], out["state"] = False, "failed"
+        out["error"] = type(exc).__name__
+    finally:
+        out["seconds"] = round(_time.perf_counter() - started, 4)
+    return out
