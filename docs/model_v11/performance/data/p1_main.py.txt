def rss_kb(pid):

        for line in open(f"/proc/{pid}/status"):

def one_rep(variant, rep):
    me._worker_warmup_probe = p1probe.warmup_probe_with_native if variant == "p1" else p1probe.base_probe
    ex = me.ModelExecutor(me.ModelExecutionConfig(concurrency=2, mode="process")); me._EXECUTOR = ex
    t = time.perf_counter(); st = ex.warm_up(); warm_s = time.perf_counter() - t
    pids = st["warm_pids"]; rec = {"variant": variant, "rep": rep, "warm_s": warm_s, "warm": st["warm"], "workers": st["warm_workers"], "pids": pids,
                                   "rss_after_warm_kb": {p: rss_kb(p) for p in pids}, "runs": []}
    seen = set()
    for k in range(4):
        t = time.perf_counter()
        r = ex.run_process_sync(bf.Timed(na.activated(run_project)), *box["args"], project_inputs_override=pert(1 + rep * 10 + k + (500 if variant == "p1" else 0)))

        digests.add((k, hashlib.sha256(pickle.dumps(value, protocol=5)).hexdigest())) if False else None
        rec["runs"].append({"seq": k, "pid": r["pid"], "first_for_pid": r["pid"] not in seen, "inner": r["inner_wall"], "roundtrip": rt,
                            "v11_models": None if not hasattr(v, "stage3_stats") or v.stage3_stats is None else v.stage3_stats["native_models"]})
        seen.add(r["pid"])
    rec["rss_after_runs_kb"] = {p: rss_kb(p) for p in pids}
